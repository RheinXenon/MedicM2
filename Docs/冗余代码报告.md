**建议**: 创建一个通用的工具函数
```python
# utils/__init__.py 或新建 utils/json_parser.py
def parse_json_from_llm_response(response: str, default_structure: Dict) -> Dict:
    """从LLM响应中提取并解析JSON"""
    import re
    import json
    
    try:
        json_match = re.search(r'\{.*\}', response, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
        else:
            return default_structure
    except Exception as e:
        print(f"JSON解析失败: {e}")
        return default_structure
```
